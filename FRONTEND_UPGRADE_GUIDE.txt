### 4. Add API Functions (Line ~150, after handleCreateCase)
`	ypescript
const handleTranscribe = async (options: any) => {
  if (!selectedTaskId) return;
  
  try {
    const response = await fetch(\\/transcribe/\\, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        enable_diarization: options.enableDiarization,
        diarization_method: options.diarizationMethod || 'pyannote',
        language: options.language || 'vi',
        fast_mode: true,
        async_mode: true
      })
    });
    
    if (!response.ok) throw new Error('Failed');
    const result = await response.json();
    
    // Update status
    setFiles(prev => prev.map(f => 
      f.task_id === selectedTaskId 
        ? { ...f, status: 'processing', celery_task_id: result.celery_task_id }
        : f
    ));
    
    // Start polling
    startStatusPolling(selectedTaskId);
    
    setTranscribeDialogOpen(false);
    setSnackbar({open: true, message: 'Transcription started', severity: 'info'});
  } catch (error) {
    setSnackbar({open: true, message: 'Failed to start', severity: 'error'});
  }
};

const handleSummarize = async (options: any) => {
  if (!selectedTaskId) return;
  
  try {
    const response = await fetch(\\/summarize/\\, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model_name: options.modelName,
        summary_type: options.summaryType || 'detailed',
        include_context: true,
        async_mode: true
      })
    });
    
    if (!response.ok) throw new Error('Failed');
    const result = await response.json();
    
    setFiles(prev => prev.map(f => 
      f.task_id === selectedTaskId 
        ? { ...f, status: 'summarizing' }
        : f
    ));
    
    startStatusPolling(selectedTaskId);
    setSummarizeDialogOpen(false);
    setSnackbar({open: true, message: 'Summarization started', severity: 'info'});
  } catch (error) {
    setSnackbar({open: true, message: 'Failed to start', severity: 'error'});
  }
};

const startStatusPolling = (taskId: string) => {
  // Clear existing
  const existing = pollingIntervals.get(taskId);
  if (existing) clearInterval(existing);
  
  // New interval
  const intervalId = setInterval(async () => {
    try {
      const response = await fetch(\\/tasks/\/status\);
      const status = await response.json();
      
      // Update state
      setFiles(prev => prev.map(f => 
        f.task_id === taskId ? { ...f, ...status } : f
      ));
      
      // Stop if done
      if (['transcribed', 'summarized', 'failed'].includes(status.status)) {
        clearInterval(intervalId);
        setPollingIntervals(prev => {
          const m = new Map(prev);
          m.delete(taskId);
          return m;
        });
        
        if (status.status === 'transcribed') {
          setSnackbar({open: true, message: 'Transcription complete!', severity: 'success'});
        } else if (status.status === 'summarized') {
          setSnackbar({open: true, message: 'Summary complete!', severity: 'success'});
        } else if (status.status === 'failed') {
          setSnackbar({open: true, message: \Failed: \\, severity: 'error'});
        }
      }
    } catch (error) {
      console.error('Polling error:', error);
    }
  }, 2000); // Every 2s
  
  setPollingIntervals(prev => new Map(prev).set(taskId, intervalId));
};
`
